`packages/ecs/index.ts`：
```typescript
export class World {
  // 现有 query 不变
  query<T>(...tokens: Token<T>[]): Entity[] { /* ... */ }
  
  // 新增 bulkQuery
  bulkQuery<T>(...tokens: Token<T>[]): BulkQueryIterator {
    const indices = this.computeIntersection(...tokens)
    const buffer = new Array<Entity>(indices.length)
    for (let i = 0; i < indices.length; i++) {
      buffer[i] = this.handles[indices[i]]!
    }
    return new BulkQueryIteratorImpl(buffer, this, tokens)
  }
}

class BulkQueryIteratorImpl implements BulkQueryIterator {
  readonly buffer: Entity[]
  private cancelled = false
  
  async *[Symbol.asyncIterator](): AsyncIterator<Entity[]> {
    // 检测改动，每迭代步检查一次 isAlive
    yield this.buffer
    if (this.cancelled) return
    // 后续步：重新计算交集并验证（简化实现）
  }
  
  return() { this.cancelled = true }
}
```

- [ ] **Step 1: 实现采集器**

`packages/app/runtime/performance-monitor.ts`：
```typescript
export class PerformanceMonitor {
  private frameCount = 0
  private lastTime = performance.now()
  private currentMetrics = createMetrics()
  
  recordQuery() { this.currentMetrics.queries++ }
  recordCollisionPair() { this.currentMetrics.collision++ }
  recordDrawCall() { this.currentMetrics.drawCalls++ }
  
  // 每帧末尾调用
  finishFrame(): PerformanceMetrics {
    const now = performance.now()
    const elapsed = now - this.lastTime
    this.frameCount++
    
    if (elapsed >= 1000) {
      this.currentMetrics.fps = this.frameCount / (elapsed / 1000)
      this.frameCount = 0
      this.lastTime = now
      
      if (performance.memory) {
        this.currentMetrics.memory = {
          heap: performance.memory.usedJSHeapSize,
          external: performance.memory.externalMemoryUsage,
        }
      }
      
      const result = { ...this.currentMetrics }
      this.currentMetrics = createMetrics()
      return result
    }
    
    return this.currentMetrics
  }
}
```

export class TransformCache {
  private cache = new Map<Entity, Matrix3x2>()
  private dirty = new Set<Entity>()
  
  invalidate(entity: Entity) {
    this.dirty.add(entity)
    this.cache.delete(entity)
  }
  
  get(entity: Entity, world: World): Matrix3x2 {
    if (this.cache.has(entity)) {
      return this.cache.get(entity)!
    }
    
    const matrix = computeTransform(entity, world)
    this.cache.set(entity, matrix)
    return matrix
  }
  
  // 组件改动时调用
  onComponentChange(entity: Entity) {
    this.invalidate(entity)
    // 也需要检查依赖此实体的子对象
    // （如果有 parent-child Transform 关系）
  }
}

export class ActionSampler {
  private frozen: ReadonlyData<Actions> | null = null
  
  read(): ReadonlyData<Actions> {
    if (this.frozen) return this.frozen
    this.frozen = Object.freeze({ ...this.current })
    return this.frozen
  }
  
  invalidate() { this.frozen = null }
}

class SpatialHash {
  private cellSize: number
  private cells = new Map<string, Entity[]>()
  
  insert(bounds: Bounds) {
    const key = `${Math.floor(bounds.minX / this.cellSize)},${Math.floor(bounds.minY / this.cellSize)}`
    if (!this.cells.has(key)) this.cells.set(key, [])
    this.cells.get(key)!.push(bounds.entity)
  }
  
  queryCell(point: Point): Entity[] {
    const key = `${Math.floor(point.x / this.cellSize)},${Math.floor(point.y / this.cellSize)}`
    return this.cells.get(key) || []
  }
}
